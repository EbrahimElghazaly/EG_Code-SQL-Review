CREATE DATABASE DDL
الشرح: تستخدم لإنشاء قاعدة بيانات جديدة. قبل إنشاء أي جدول، لازم يكون عندك قاعدة بيانات.
مثال عملي: لو عايز تعمل نظام لإدارة مدرسة، هتعمل قاعدة بيانات باسم "SchoolDB".
CREATE DATABASE SchoolDB; ✅ تم إنشاء قاعدة البيانات "SchoolDB"
DDL إنشاء
CREATE TABLE DDL
الشرح: تنشئ جدول جديد داخل قاعدة البيانات. تحدد اسم الجدول، الأعمدة، ونوع كل عمود.
مثال عملي: هنعمل جدول "Students" فيه بيانات الطلاب: ID، الاسم، العمر.
CREATE TABLE Students ( ID INT PRIMARY KEY, Name VARCHAR(50) NOT NULL, Age INT ); ✅ تم إنشاء جدول Students
DDL هيكل
INSERT DML
الشرح: إضافة بيانات جديدة (سجلات) إلى الجدول. لازم تطابق الأعمدة مع القيم.
مثال عملي: هنضيف طالب جديد اسمه "أحمد" عمره 22 سنة.
INSERT INTO Students (ID, Name, Age) VALUES (1, 'أحمد محمد', 22); ✅ تم إدراج الطالب أحمد
DML إدراج
UPDATE DML
الشرح: تعديل البيانات الموجودة. لازم تستخدم WHERE عشان تحدد السجل المطلوب، وإلا هتعدل كل البيانات!
مثال عملي: هنغير عمر الطالب اللي رقمه 1 من 22 لـ 23.
UPDATE Students SET Age = 23 WHERE ID = 1; ✅ تم تحديث عمر الطالب رقم 1
DML تعديل
DELETE DML
الشرح: حذف سجلات من الجدول. لازم تستخدم WHERE عشان تحدد السجل، وإلا هتحذف كل البيانات!
مثال عملي: هنحذف الطالب اللي رقمه 1.
DELETE FROM Students WHERE ID = 1; ✅ تم حذف الطالب رقم 1
DML حذف
SELECT DQL
الشرح: استعلام لجلب البيانات من الجدول. تقدر تختار أعمدة معينة أو كل الأعمدة (*).
مثال عملي: هنعرض أسماء وأعمار جميع الطلاب.
SELECT Name, Age FROM Students; ✅ عرض أسماء وأعمار الطلاب
DQL استعلام
WHERE
الشرح: تصفية النتائج بحيث ترجع فقط السجلات اللي تحقق الشرط المطلوب.
مثال عملي: نعرض الطلاب اللي أعمارهم أكبر من 18 سنة والقسم بتاعهم CS.
SELECT * FROM Students WHERE Age > 18 AND Dept = 'CS'; ✅ طلاب CS أعمارهم أكبر من 18
تصفية
ORDER BY
الشرح: ترتيب النتائج تصاعدياً (ASC) أو تنازلياً (DESC).
مثال عملي: نعرض الطلاب مرتبين تنازلياً حسب العمر.
SELECT * FROM Students ORDER BY Age DESC; ✅ الطلاب من الأكبر للأصغر
ترتيب
GROUP BY
الشرح: تجميع البيانات حسب عمود معين، عادة مع دوال التجميع (COUNT, SUM, AVG).
مثال عملي: عدد الطلاب في كل قسم.
SELECT Dept, COUNT(*) AS StudentCount FROM Students GROUP BY Dept; ✅ عدد الطلاب في كل قسم
تجميع
HAVING
الشرح: تصفية بعد التجميع (مع GROUP BY). الفرق عن WHERE إن WHERE قبل التجميع وHAVING بعده.
مثال عملي: الأقسام اللي فيها أكثر من 5 طلاب.
SELECT Dept, COUNT(*) AS Count FROM Students GROUP BY Dept HAVING COUNT(*) > 5; ✅ الأقسام اللي فيها أكثر من 5 طلاب
تصفية تجميع
JOIN
الشرح: ربط جدولين مع بعض. أنواعه: INNER (المتطابق فقط)، LEFT (كل الجدول الأيسر)، RIGHT، FULL.
مثال عملي: عرض اسم الطالب مع اسم الكورس اللي مسجل فيه.
SELECT S.Name, C.Title FROM Students S INNER JOIN Courses C ON S.ID = C.StudentID; ✅ كل طالب مع الكورس المسجل فيه
ربط
PRIMARY KEY
الشرح: مفتاح أساسي يميز كل سجل بشكل فريد. لا يسمح بتكرار القيم أو NULL.
مثال عملي: كل طالب له ID فريد يميزه عن غيره.
CREATE TABLE Employees ( EmpID INT PRIMARY KEY, Name VARCHAR(50) ); ✅ EmpID هو المفتاح الأساسي
قيد فريد
FOREIGN KEY
الشرح: مفتاح خارجي يربط جدول بجدول آخر. بيشير للمفتاح الأساسي في الجدول التاني.
مثال عملي: جدول الطلبات مرتبط بجدول العملاء عن طريق CustomerID.
CREATE TABLE Orders ( OrderID INT PRIMARY KEY, CustomerID INT, FOREIGN KEY (CustomerID) REFERENCES Customers(ID) ); ✅ Orders مرتبط بـ Customers
قيد علاقة
CONSTRAINT
الشرح: قيد يمكن تطبيقه على 3 مستويات: مستوى العمود، مستوى الجدول، أو بعد الإنشاء بـ ALTER.
مثال عملي: نفس القيد لكن بطرق مختلفة.
-- مستوى العمود ID INT PRIMARY KEY -- مستوى الجدول CONSTRAINT PK_Emp PRIMARY KEY (ID) -- بعد الإنشاء (ALTER) ALTER TABLE Employees ADD CONSTRAINT UQ_Email UNIQUE (Email);
قيد مستويات
DEFAULT
الشرح: قيمة افتراضية للعمود لو مفيش قيمة اتحطت عند الإدراج.
مثال عملي: لو مفيش حالة اتحطت، يبقى افتراضي "نشط".
CREATE TABLE Users ( ID INT PRIMARY KEY, Status VARCHAR(20) DEFAULT 'نشط' ); ✅ لو مفيش قيمة، يبقى Status = 'نشط'
قيد افتراضي
UNIQUE
الشرح: يمنع تكرار القيم في العمود. بيضمن إن كل قيمة فريدة.
مثال عملي: البريد الإلكتروني ما يتكررش.
CREATE TABLE Users ( Email VARCHAR(100) UNIQUE ); ✅ يمنع إدخال بريد مكرر
قيد فريد
CHECK
الشرح: يتحقق من شرط معين قبل إدخال أو تعديل البيانات.
مثال عملي: مينفعش الراتب يكون أقل من صفر.
CREATE TABLE Employees ( Salary DECIMAL(10,2) CHECK (Salary >= 0) ); ✅ يمنع إدخال راتب سالب
قيد تحقق
LIKE
الشرح: البحث بنمط معين في النصوص. % = أي عدد من الحروف، _ = حرف واحد.
مثال عملي: البحث عن طلاب أسماؤهم تبدأ بحرف الألف.
-- تبدأ بـ 'أ' WHERE Name LIKE 'أ%' -- تحتوي على 'مد' WHERE Name LIKE '%مد%' -- ثاني حرف 'ح' WHERE Name LIKE '_ح%'
نمط
BETWEEN
الشرح: البحث عن القيم في نطاق محدد (يشمل الحدين).
مثال عملي: الطلاب اللي أعمارهم بين 18 و 25 سنة.
SELECT * FROM Students WHERE Age BETWEEN 18 AND 25; ✅ الطلاب من 18 لـ 25 سنة
نطاق
IN
الشرح: التحقق من وجود القيمة في قائمة محددة.
مثال عملي: الطلاب في أقسام CS أو IS.
SELECT * FROM Students WHERE Dept IN ('CS', 'IS'); ✅ طلاب CS أو IS
قائمة
CASE
الشرح: منطق شرطي زي IF-ELSE داخل الاستعلام. بيصنف البيانات حسب شروط.
مثال عملي: تصنيف الطلاب حسب العمر: كبير، متوسط، صغير.
SELECT Name, CASE WHEN Age >= 25 THEN 'كبير' WHEN Age >= 18 THEN 'متوسط' ELSE 'صغير' END AS Category FROM Students; ✅ كل طالب مع تصنيفه العمري
شرطي
SUBQUERY
الشرح: استعلام جوه استعلام تاني. الاستعلام الداخلي بيتنفذ الأول ونتيجته بتتستخدم في الاستعلام الخارجي.
مثال عملي: الطلاب اللي في أقسام موجودة في القاهرة.
SELECT Name FROM Students WHERE Dept IN ( SELECT DeptID FROM Departments WHERE Location = 'القاهرة' ); ✅ الطلاب في أقسام القاهرة
متداخل
Agg Functions
الشرح: دوال بتتعامل مع مجموعة من السجلات وترجع قيمة واحدة: COUNT (عدد), SUM (مجموع), AVG (متوسط), MAX (أكبر), MIN (أصغر).
مثال عملي: إحصائيات عن أعمار الطلاب.
SELECT COUNT(*) AS TotalStudents, AVG(Age) AS AverageAge, MAX(Age) AS Oldest, MIN(Age) AS Youngest FROM Students; ✅ إحصائيات عن الطلاب
تجميع
VIEW
الشرح: جدول افتراضي مبني على استعلام. بيسهل الوصول للبيانات من غير ما تكتب الاستعلام كل مرة.
مثال عملي: عرض افتراضي للطلاب الكبار (18+).
CREATE VIEW AdultStudents AS SELECT Name, Age FROM Students WHERE Age >= 18; ✅ استخدمها: SELECT * FROM AdultStudents
عرض
SQL SERVER
الشرح: نظام إدارة قواعد بيانات من Microsoft. فيه ميزات إضافية زي TOP, GETDATE(), @@ROWCOUNT.
مثال عملي: جلب أول 10 طلاب، التاريخ الحالي، وعدد السجلات المتأثرة.
SELECT TOP 10 * FROM Students; SELECT GETDATE() AS CurrentDateTime; SELECT @@ROWCOUNT AS RowsAffected; ✅ ميزات SQL Server
